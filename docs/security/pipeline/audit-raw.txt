  schedule:

  workflow_dispatch:

  contents: read
  issues: write

  NODE_VERSION: '22'

jobs:
  health-check:
    name: Daily Health Check
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'

      - name: Install Dependencies
        run: npm ci

      - name: Run ESLint
        run: npm run lint

      - name: TypeScript Check
        run: npx tsc --noEmit

      - name: Security Audit
        run: npm audit --audit-level moderate || echo "Security audit completed with warnings"

      - name: Build Test
        run: npm run build

      - name: Run Tests
        run: npm test

      - name: Generate Health Report
        if: always()
        run: |
          echo "## Daily Health Check Report" >> $GITHUB_STEP_SUMMARY

=== monitoring-health.yml ===
name: Monitoring and Health Checks

on:
  schedule:
    - cron: '0 */6 * * *'
  workflow_dispatch:

permissions:
  contents: read
  issues: write

env:
  PRODUCTION_URL: 'https://namias.tech'
  BLOG_MONITOR_ENABLED: 'false'

jobs:
  uptime-check:
    name: Uptime and Availability Check
    runs-on: ubuntu-latest
    steps:
      - name: Check Website Availability
        run: |
          echo "## Uptime Check Results" >> $GITHUB_STEP_SUMMARY
          echo "**Date**: $(date)" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY

          FAILED=0

          # Check main site
          if curl -f -s --max-time 30 "${{ env.PRODUCTION_URL }}" > /dev/null; then
            echo "**Main Site**: Available" >> $GITHUB_STEP_SUMMARY
          else
            echo "**Main Site**: Unavailable" >> $GITHUB_STEP_SUMMARY
            FAILED=1
          fi

          # Check chat service health endpoint
          CHAT_HEALTH_STATUS=$(curl -sS --max-time 20 -o /tmp/chat-health.json -w "%{http_code}" "${{ env.PRODUCTION_URL }}/api/chat")
          if [ "$CHAT_HEALTH_STATUS" -eq 200 ] && grep -q '"status":"active"' /tmp/chat-health.json; then
            echo "**Chat Health**: Active" >> $GITHUB_STEP_SUMMARY
          else
            echo "**Chat Health**: Inactive or unreachable" >> $GITHUB_STEP_SUMMARY
            FAILED=1
          fi

          # Synthetic chatbot probe (ensures non-empty message contract)
          CHAT_PROBE_RESPONSE=$(curl -sS --max-time 30 -X POST "${{ env.PRODUCTION_URL }}/api/chat" \
            -H "Content-Type: application/json" \
            -d '{"message":"Hello Keneth AI"}' \
            -w "HTTPSTATUS:%{http_code}")

          CHAT_PROBE_BODY=$(echo "$CHAT_PROBE_RESPONSE" | sed -e 's/HTTPSTATUS:.*//g')
          CHAT_PROBE_STATUS=$(echo "$CHAT_PROBE_RESPONSE" | tr -d '\n' | sed -e 's/.*HTTPSTATUS://')

          if [ "$CHAT_PROBE_STATUS" -eq 200 ] && echo "$CHAT_PROBE_BODY" | grep -q '"message"'; then
            echo "**Chat Probe**: Passed" >> $GITHUB_STEP_SUMMARY
          else
            echo "**Chat Probe**: Failed" >> $GITHUB_STEP_SUMMARY
            FAILED=1
          fi

          # Check blog page only when blog route is publicly enabled
          if [ "${{ env.BLOG_MONITOR_ENABLED }}" = "true" ]; then
            if curl -f -s --max-time 15 "${{ env.PRODUCTION_URL }}/blog" > /dev/null; then
              echo "**/blog**: Available" >> $GITHUB_STEP_SUMMARY
            else
              echo "**/blog**: Unavailable" >> $GITHUB_STEP_SUMMARY
              FAILED=1
            fi
          else
            echo "**/blog**: Skipped (feature currently disabled)" >> $GITHUB_STEP_SUMMARY
          fi

          if [ "$FAILED" -eq 1 ]; then
            exit 1
          fi

      - name: Create Alert on Downtime
        if: failure()
        uses: actions/github-script@v7
        with:
          github-token: ${{ secrets.GITHUB_TOKEN }}
          script: |
            const title = `Website Downtime Alert - ${new Date().toDateString()}`;
            const body = `## Website Downtime Detected

            **Alert Time**: ${new Date().toISOString()}
            **URL**: ${{ env.PRODUCTION_URL }}
            **Blog Monitor Enabled**: ${{ env.BLOG_MONITOR_ENABLED }}
            **Chat Endpoint**: ${{ env.PRODUCTION_URL }}/api/chat

            Please check hosting provider status and server logs.

            *Automatically generated by the monitoring system.*`;

            github.rest.issues.create({
              owner: context.repo.owner,
              repo: context.repo.repo,
              title: title,
              body: body,
              labels: ['alert', 'downtime', 'high-priority']
            });

  ssl-certificate-check:
    name: SSL Certificate Check
    runs-on: ubuntu-latest
    continue-on-error: true
    steps:
      - name: Check SSL Certificate
        run: |
          DOMAIN=$(echo "${{ env.PRODUCTION_URL }}" | sed 's|https\?://||' | cut -d'/' -f1)

          echo "## SSL Certificate Status" >> $GITHUB_STEP_SUMMARY
          echo "**Domain**: $DOMAIN" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY

          if CERT_INFO=$(echo | openssl s_client -servername "$DOMAIN" -connect "$DOMAIN:443" 2>/dev/null | openssl x509 -noout -dates 2>/dev/null); then
            EXPIRY=$(echo "$CERT_INFO" | grep notAfter | cut -d'=' -f2)
            if [ -n "$EXPIRY" ]; then
              EXPIRY_TS=$(date -d "$EXPIRY" +%s 2>/dev/null || echo "0")
              NOW_TS=$(date +%s)
              DAYS=$(( (EXPIRY_TS - NOW_TS) / 86400 ))
              echo "**Expiry**: $EXPIRY" >> $GITHUB_STEP_SUMMARY
              echo "**Days Until Expiry**: $DAYS" >> $GITHUB_STEP_SUMMARY
              if [ "$DAYS" -lt 30 ]; then
                echo "**Certificate expiring soon**" >> $GITHUB_STEP_SUMMARY
              else
                echo "**Certificate valid**" >> $GITHUB_STEP_SUMMARY
              fi
            fi
          else
            echo "Could not retrieve certificate info" >> $GITHUB_STEP_SUMMARY
          fi

  monitoring-summary:
    name: Monitoring Summary
    runs-on: ubuntu-latest
    needs: [uptime-check, ssl-certificate-check]
    if: always()
    steps:
      - name: Generate Summary
        run: |
          echo "## System Health Dashboard" >> $GITHUB_STEP_SUMMARY
          echo "**Report**: $(date)" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "| Service | Status |" >> $GITHUB_STEP_SUMMARY
          echo "|---------|--------|" >> $GITHUB_STEP_SUMMARY
          echo "| Uptime | ${{ needs.uptime-check.result == 'success' && 'Online' || 'Issues' }} |" >> $GITHUB_STEP_SUMMARY
          echo "| SSL | ${{ needs.ssl-certificate-check.result == 'success' && 'Valid' || 'Check' }} |" >> $GITHUB_STEP_SUMMARY

=== pentestagent-pr-check.yml ===
name: PentestAgent PR Security Check

on:
  pull_request:
    branches: [main]

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  security-pr-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Check for security-sensitive file changes
        id: diff-check
        run: |
          CHANGED_FILES=$(git diff --name-only origin/main...HEAD)
          echo "Changed files:"
          echo "$CHANGED_FILES"

          SECURITY_PATTERNS=("api/" ".env" "next.config" "middleware" "package.json" "eslint.config")
          HAS_SECURITY_CHANGES=false

          for pattern in "${SECURITY_PATTERNS[@]}"; do
            if echo "$CHANGED_FILES" | grep -qi "$pattern"; then
              HAS_SECURITY_CHANGES=true
              echo "Security-relevant file changed: matches pattern '$pattern'"
            fi
          done

          echo "has_security_changes=$HAS_SECURITY_CHANGES" >> $GITHUB_OUTPUT

      - name: Setup Python 3.11
        if: ${{ steps.diff-check.outputs.has_security_changes == 'true' }}
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Clone and install PentestAgent (lightweight)
        if: ${{ steps.diff-check.outputs.has_security_changes == 'true' }}
        run: |
          git clone https://github.com/GH05TCREW/pentestagent.git pentestagent
          cd pentestagent
          pip install -e ".[all]"

      - name: Run lightweight security scan
        if: ${{ steps.diff-check.outputs.has_security_changes == 'true' }}
        id: scan
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
        continue-on-error: true
        run: |
          if [ -z "$ANTHROPIC_API_KEY" ] && [ -z "$OPENAI_API_KEY" ]; then
            echo "::notice::No LLM API key configured; running static security header verification only."
            mkdir -p docs/security/pentestagent/findings
            OUT="docs/security/pentestagent/findings/pr-security-check-${{ github.event.number }}.json"
            node scripts/static-security-check.mjs > "$OUT" 2>&1 || true
            if [ ! -s "$OUT" ]; then
              echo '{"summary":{"errorCount":0,"warningCount":0,"note":"static-check-produced-no-output"}}' > "$OUT"
            fi
            echo "report=$OUT" >> $GITHUB_OUTPUT
            exit 0
          fi

          OUT="docs/security/pentestagent/findings/pr-security-check-${{ github.event.number }}.json"
          cd pentestagent
          pentestagent run -t https://namias.tech \
            --playbook ../docs/security/pentestagent/playbooks/header_security.yaml \
            --output "../$OUT" \
            --json || true
          echo "report=$OUT" >> $GITHUB_OUTPUT

      - name: Upload PR scan results
        if: ${{ steps.diff-check.outputs.has_security_changes == 'true' }}
        uses: actions/upload-artifact@v4
        with:
          name: pentestagent-pr-check-${{ github.event.number }}
          path: docs/security/pentestagent/findings/
          retention-days: 30
          if-no-files-found: ignore

      - name: Comment security check result
        if: ${{ steps.diff-check.outputs.has_security_changes == 'true' }}
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs');
            const path = '${{ steps.scan.outputs.report }}';
            let findings = { summary: { errorCount: 0, warningCount: 0 } };
            try {
              if (path && fs.existsSync(path)) {
                findings = JSON.parse(fs.readFileSync(path, 'utf8'));
              }
            } catch (e) {
              console.log('Could not parse findings file, using defaults');
            }
            const hasIssues = findings && findings.summary && findings.summary.errorCount > 0;

            let body = '## âœ… PentestAgent Security Check\n\n';
            if (hasIssues) {
              body = '## âš ï¸ PentestAgent Security Check\n\n';
              body += '**Security issues were detected in this PR.**\n\n';
              body += `- Errors: ${findings.summary.errorCount}\n`;
              body += `- Warnings: ${findings.summary.warningCount}\n\n`;
              body += 'Please review the findings and address any security concerns.\n';
            } else {
              body += 'No security issues detected. The security headers and configuration look good.\n';
              if (findings && findings.summary && findings.summary.note) {
                body += `\n_Note: ${findings.summary.note}_\n`;
              }
            }
            body += '\n---\n';
            body += '_This check runs when security-relevant files are changed._';

            try {
              await github.rest.issues.createComment({
                issue_number: context.issue.number,
                owner: context.repo.owner,
                repo: context.repo.repo,
                body: body
              });
            } catch (e) {
              console.log('Could not post comment:', e.message);
            }

              owner,
              repo,
              issue_number: issueNumber,
              per_page: 100,
